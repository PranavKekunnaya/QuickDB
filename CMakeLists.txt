cmake_minimum_required(VERSION 3.16)
project(QuickDB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(quickdb src/main.cpp)
target_compile_options(quickdb PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_database.py $<TARGET_FILE:quickdb>)
